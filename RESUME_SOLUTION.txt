  2. UTILISER EN PYTHON

     # Tester une combinaison
     GET http://localhost:8001/api/journal/combination/34/38
     
     # Générer un journal
     POST http://localhost:8001/api/journal/generate
     {"numbers": [34, 38, 12, 45]}
     
     # Valider un univers
     POST http://localhost:8001/api/journal/validate-universe
     {"numbers": [34, 38], "universe": "mundo"}
